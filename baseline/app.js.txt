const LIFT_CM_PER_REV = 0.35;
const LIFT_PULSES_PER_REV = 10000;
const DEFAULT_LIFT_TRAVEL_CM = 180;

function liftTravelCm() {
  const fromConfig = Number(state.config?.lift_travel_height);
  if (Number.isFinite(fromConfig) && fromConfig > 0) return fromConfig;
  return DEFAULT_LIFT_TRAVEL_CM;
}

function liftTravelMeters() {
  return liftTravelCm() / 100;
}

function liftBaseHeightCm() {
  const raw = state.config?.lift_base_height_cm;
  if (raw === null || raw === undefined || raw === "") return null;
  const value = Number(raw);
  return Number.isFinite(value) && value >= 0 ? value : null;
}

function liftHeightScale() {
  const travel = liftTravelCm();
  const base = liftBaseHeightCm();
  const hasBase = base !== null;
  return {
    travel,
    base: hasBase ? base : 0,
    hasBase,
    min: hasBase ? base : 0,
    max: hasBase ? base + travel : travel,
  };
}

function liftSliderValueToExtensionCm(sliderValue) {
  const scale = liftHeightScale();
  return Number(sliderValue) - (scale.hasBase ? scale.base : 0);
}

function liftExtensionToSliderValue(extensionCm) {
  const scale = liftHeightScale();
  const ext = Math.max(0, Math.min(scale.travel, Number(extensionCm) || 0));
  return scale.hasBase ? scale.base + ext : ext;
}

function liftCmToHeightPercent(cm) {
  const travel = liftTravelCm();
  if (!(travel > 0)) return 0;
  // 整数百分比一步是 travel/100（180 cm 行程约 1.8 cm）。10 cm 四舍五入会变成 6%→10.8 cm。
  const pct = (Number(cm) / travel) * 100;
  return Math.max(0, Math.min(Number(pct.toFixed(4)), 100));
}

function liftMetersToHeightPercent(meters) {
  return liftCmToHeightPercent(Number(meters) * 100);
}

function readLiftTargetCm(sliderId) {
  return Number(document.getElementById(sliderId)?.value || 0);
}

async function sendLiftGotoFromSlider(sliderId) {
  const extension = liftSliderValueToExtensionCm(readLiftTargetCm(sliderId));
  const height = liftCmToHeightPercent(extension);
  state.lastRemoteLiftTargetPct = height;
  return sendLiftCommand({ action: "goto", height });
}

function liftCommandSpeedCmS(ratio = Number(document.getElementById("roofLiftSpeedInput")?.value
  ?? document.getElementById("liftSpeedInput")?.value
  ?? 0.3)) {
  const maxPr = Number(state.config?.lift_max_pv_speed_pr_s ?? 100000);
  const minPr = Number(state.config?.lift_min_pv_speed_pr_s ?? 3000);
  const pr = Math.max(minPr, Math.min(maxPr, Math.round(Number(ratio) * maxPr)));
  return (pr / LIFT_PULSES_PER_REV) * LIFT_CM_PER_REV;
}

function formatLiftSpeedLabel(ratio) {
  const value = Number(ratio || 0);
  return `${value.toFixed(2)} · ${liftCommandSpeedCmS(value).toFixed(2)} cm/s`;
}

function applyLiftTravelToControls() {
  const scale = liftHeightScale();
  ["roofLiftHeightSlider", "liftHeightSlider"].forEach((id) => {
    const el = document.getElementById(id);
    if (!el) return;
    const previous = Number(el.value || 0);
    const prevMin = Number(el.min || 0);
    const prevMax = Number(el.max || 0);
    const wasMetersScale = prevMax <= 3;
    const prevWasTotal = el.dataset.liftScale === "total" || prevMin > 0;
    let extension = previous;
    if (wasMetersScale) extension = previous * 100;
    else if (prevWasTotal) extension = previous - prevMin;
    const minChanged = el.min !== String(scale.min);
    const maxChanged = el.max !== String(scale.max);
    el.min = String(scale.min);
    el.max = String(scale.max);
    el.step = "1";
    el.dataset.liftScale = scale.hasBase ? "total" : "extension";
    // 拖动中或量程未变时不要写 value，否则浏览器会丢掉 range 的原生拖动。
    if (state.liftTargetDragging || (!minChanged && !maxChanged && !wasMetersScale)) return;
    el.value = String(liftExtensionToSliderValue(extension));
  });
  ["roofLiftCurrentHeight", "liftCurrentHeight"].forEach((id) => {
    const el = document.getElementById(id);
    if (!el) return;
    el.min = String(scale.min);
    el.max = String(scale.max);
    el.step = "0.1";
  });
  syncLiftTargetLabels();
  updateLiftSpeedLabels();
}

function syncLiftTargetLabels() {
  const pairs = [
    ["roofLiftHeightSlider", "roofLiftHeightValue"],
    ["liftHeightSlider", "liftHeightValue"],
  ];
  pairs.forEach(([sliderId, outputId]) => {
    const slider = document.getElementById(sliderId);
    const output = document.getElementById(outputId);
    if (!slider || !output) return;
    const extension = liftSliderValueToExtensionCm(slider.value);
    const total = liftHeightScale().hasBase ? Number(slider.value) : null;
    output.textContent = formatLiftHeightLabel(extension, total);
  });
}

function updateLiftSpeedLabels(ratio) {
  const value = ratio === undefined
    ? Number(document.getElementById("roofLiftSpeedInput")?.value
      ?? document.getElementById("liftSpeedInput")?.value
      ?? 0.3)
    : Number(ratio);
  const label = formatLiftSpeedLabel(value);
  const roofOut = document.getElementById("roofLiftSpeedValue");
  const drawerOut = document.getElementById("liftSpeedValue");
  if (roofOut) roofOut.textContent = label;
  if (drawerOut) drawerOut.textContent = label;
}

function formatLiftHeightLabel(extensionCm, totalCm = null) {
  if (extensionCm === null || extensionCm === undefined || !Number.isFinite(Number(extensionCm))) {
    return "未知";
  }
  const extend = Number(extensionCm);
  // null/undefined 不能用 Number() 判断：Number(null)===0，会误显示「总高 0.0」。
  if (totalCm !== null && totalCm !== undefined && totalCm !== "" && Number.isFinite(Number(totalCm))) {
    return `伸出 ${extend.toFixed(1)} · 总高 ${Number(totalCm).toFixed(1)} cm`;
  }
  return `伸出 ${extend.toFixed(1)} cm（总高待配车厢离地）`;
}

function updateLiftCurrentHeightUi(heightCm, totalHeightCm = null) {
  const travel = liftTravelCm();
  const value = Number.isFinite(Number(heightCm)) ? Math.max(0, Math.min(travel, Number(heightCm))) : null;
  const base = liftBaseHeightCm();
  const hasTotal = totalHeightCm !== null && totalHeightCm !== undefined && totalHeightCm !== ""
    && Number.isFinite(Number(totalHeightCm));
  const total = hasTotal
    ? Number(totalHeightCm)
    : (value !== null && base !== null ? base + value : null);
  [
    ["roofLiftCurrentHeight", "roofLiftCurrentHeightValue"],
    ["liftCurrentHeight", "liftCurrentHeightValue"],
  ].forEach(([sliderId, outputId]) => {
    const slider = document.getElementById(sliderId);
    const output = document.getElementById(outputId);
    if (slider) {
      // 滑条量程用“总高上限”更直观：车厢离地 + 行程；未配车厢离地时仍用行程。
      const span = base !== null ? base + travel : travel;
      slider.min = base !== null ? String(base) : "0";
      slider.max = String(span);
      if (total !== null) slider.value = String(total);
      else if (value !== null) slider.value = String(value);
    }
    if (output) {
      output.textContent = formatLiftHeightLabel(value, total);
    }
  });
  document.querySelectorAll('[data-lift-status-field="height"]').forEach((item) => {
    item.textContent = formatLiftHeightLabel(value, total);
  });
}

function getClientId() {
  const existing = sessionStorage.getItem("patrol_client_id");
  if (existing) return existing;

  const generated = `web-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}`;
  sessionStorage.setItem("patrol_client_id", generated);

async function loadDevices() {
  const response = await fetch("/api/devices");
  const payload = await response.json();
  state.devices = payload.devices || [];
  state.device = pickDevice(state.device?.id, (device) => device.capabilities?.includes("ptz"))
    || pickDevice("dog_ptz_1")
    || pickDevice("hikvision_ptz_1")
    || state.devices[0];
  state.videoDevice = pickDevice(state.videoDevice?.id, (device) => isSelectableVideoDevice(device))
    || pickDevice("m20_front_camera", isSelectableVideoDevice)
    || pickDevice("m20_rear_camera", isSelectableVideoDevice)
    || state.devices.find(isSelectableVideoDevice);
  state.robotDevice = pickDevice("m20_robot_1");
  state.liftDevice = pickDevice("lift_pole_1");

  renderDeviceSelectors();
  if (state.videoDevice?.video?.player_url && state.videoDevice.video.enabled !== false) {
    await syncVideoPlayers();
    await activateVideoDevice(state.videoDevice, { logSwitch: false });
    renderAuxVideoControls();
  } else {
    stopAllVideoPlayers();
    document.getElementById("deviceName").textContent = "未开启视频";
    state.auxVideoDevice = null;
    renderAuxVideoControls();
  }
  updateControlMode();
  await loadStatus();
  await loadLease();
  await loadRoofPtzStatus();
  await loadRoofPtzLease();
  await loadRobotStatus();
  await loadLiftStatus();
  await loadLiftLease();
  await loadRobotLease();
  if (state.robotDevice) {
    const robotSpeed = document.getElementById("robotSpeedInput");
    if (robotSpeed) {
      await sendRobotCommand({ action: "set_speed", speed: Number(robotSpeed.value) }).catch(() => {});
    }
  }
  log("设备已加载", state.device);
}

function pickDevice(id, predicate = null) {
  return state.devices.find((device) => {
    if (id && device.id !== id) return false;
    return predicate ? predicate(device) : true;
  });
}

function renderVideoDeviceList() {
  const target = document.getElementById("videoDeviceList");
  const videoDevices = state.devices.filter((device) => device.video && !device.legacy);
  if (!videoDevices.length) {
    target.innerHTML = "";
    return;
  }
  target.innerHTML = videoDevices.map((device) => {
    const active = state.videoLayout !== VIDEO_LAYOUT_QUAD && state.videoDevice?.id === device.id;
    const enabled = device.video.enabled !== false;
    const rawStatus = state.videoPlayers.get(videoPlayerKey(VIDEO_ROLE_MAIN, device.id))?.status || (enabled ? "idle" : "disabled");
    const status = stableStatus(`main-video-button:${device.id}`, normalizeStatus(rawStatus));
    return `
      <button type="button" data-video-device="${device.id}" data-active="${active}" data-player-status="${status}" title="${escapeHtml(device.name)}" aria-label="${escapeHtml(device.name)}" ${enabled ? "" : "disabled"}>
        ${videoDeviceIconSvg(device)}
      </button>
    `;
  }).join("") + `
    <button type="button" data-video-layout="quad" data-active="${state.videoLayout === VIDEO_LAYOUT_QUAD}" data-player-status="${quadVideoStatus()}" title="四宫格" aria-label="四宫格">
      ${quadVideoIconSvg()}
    </button>
  `;
}

function quadVideoStatus() {
  const devices = quadVideoDeviceIds().map((id) => pickDevice(id)).filter(Boolean);
  if (!devices.length) return "unknown";
  const statuses = devices.map((device) => {
    if (!isSelectableVideoDevice(device)) return "disabled";
    const player = state.videoPlayers.get(videoPlayerKey(VIDEO_ROLE_MAIN, device.id));
    return normalizeStatus(player?.status || "idle");
  });
  if (statuses.some((status) => status === "error")) return "error";
  if (statuses.every((status) => status === "ok")) return "ok";
  if (statuses.every((status) => status === "disabled")) return "disabled";
  return "connecting";
}

function quadVideoDeviceIds() {
  const m20Ids = state.operatorDirection === "rear"
    ? ["m20_rear_camera", "m20_front_camera"]
    : ["m20_front_camera", "m20_rear_camera"];
  return [...m20Ids, "dog_ptz_1", "roof_ptz_1"];
}

function normalizeStatus(value) {
  if (value === "connected") return "ok";
  if (value === "connecting" || value === "idle") return "connecting";
  if (value === "ok" || value === "error" || value === "disabled" || value === "unknown") return value;
  return value ? String(value) : "unknown";
}

function stableStatus(key, nextStatus) {
  const next = normalizeStatus(nextStatus);
  const now = Date.now();
  const entry = state.stableStatuses.get(key);
  if (!entry) {
    const initial = next === "error" ? "connecting" : next;
    state.stableStatuses.set(key, { display: initial, pending: next, pendingAt: now });
    return initial;
  }
  if (next === "ok" || next === "disabled") {
    entry.display = next;
    entry.pending = next;
    entry.pendingAt = now;
    return entry.display;
  }
  if (entry.display === next) {
    entry.pending = next;
    entry.pendingAt = now;
    return entry.display;
  }
  if (entry.pending !== next) {
    entry.pending = next;
    entry.pendingAt = now;
  }
  const delayMs = next === "error" ? 3500 : next === "connecting" ? 1200 : 2500;
  if (now - entry.pendingAt >= delayMs) {
    entry.display = next;
  }
  return entry.display;
}

function videoPlayerKey(role, deviceId) {
  return `${role}:${deviceId}`;
}

function parseVideoPlayerKey(key) {
  const index = key.indexOf(":");
  if (index < 0) return { role: VIDEO_ROLE_MAIN, deviceId: key };
  return { role: key.slice(0, index), deviceId: key.slice(index + 1) };
}

async function syncVideoPlayers() {
  const enabledDevices = selectableVideoDevices();
  const enabledIds = new Set(enabledDevices.map((device) => device.id));
  for (const key of Array.from(state.videoPlayers.keys())) {
    const { role, deviceId } = parseVideoPlayerKey(key);
    if (!enabledIds.has(deviceId)) {
      stopWebRtcPlayer(deviceId, { role, allowReconnect: false, removeElement: role === VIDEO_ROLE_MAIN });
    }
  }
  if (state.auxVideoDevice && !enabledIds.has(state.auxVideoDevice.id)) {
    state.auxVideoDevice = null;
    renderAuxVideoControls();
  }
  await Promise.all(enabledDevices.map((device) => startWebRtcPlayer(device, { role: VIDEO_ROLE_MAIN })));
  if (state.auxVideoDevice?.video?.player_url) {
    await startWebRtcPlayer(state.auxVideoDevice, { role: VIDEO_ROLE_AUX });
  }
}

async function activateVideoDevice(device, { logSwitch = true } = {}) {
  if (!device?.video?.player_url || device.video.enabled === false) return;
  state.videoLayout = VIDEO_LAYOUT_SINGLE;
  state.videoDevice = device;
  document.getElementById("videoFrameWrap").dataset.videoLayout = VIDEO_LAYOUT_SINGLE;
  document.getElementById("deviceName").textContent = device.name;
  renderVideoDeviceList();
  renderAuxVideoControls();
  updateVideoOverlayMode(device.id);
  ensureVideoElement(device, VIDEO_ROLE_MAIN);
  setActiveMainVideoElement(device.id);
  await startWebRtcPlayer(device, { role: VIDEO_ROLE_MAIN });
  setActiveMainVideoElement(device.id);
  renderQuadVideoLabels();
  if (logSwitch) {
    log("主视频已切换", { device_id: device.id, name: device.name });
  }
}

function setActiveMainVideoElement(deviceId) {
  const wrap = document.getElementById("videoFrameWrap");
  wrap.dataset.videoLayout = state.videoLayout;
  wrap.querySelectorAll('.video-frame[data-video-role="main"]').forEach((video) => {
    const isQuadItem = state.videoLayout === VIDEO_LAYOUT_QUAD && quadVideoDeviceIds().includes(video.dataset.videoDevice);
    video.dataset.active = String(isQuadItem || video.dataset.videoDevice === deviceId);
  });
}

async function activateQuadVideoLayout({ logSwitch = true } = {}) {
  const wrap = document.getElementById("videoFrameWrap");
  const referenceDevice = pickDevice("m20_front_camera", isSelectableVideoDevice) || state.videoDevice || selectableVideoDevices()[0];
  if (!referenceDevice) return;

  closeAuxVideo({ logClose: false });
  collapseOverlayDrawers();
  state.videoLayout = VIDEO_LAYOUT_QUAD;
  state.videoDevice = referenceDevice;
  document.getElementById("deviceName").textContent = "四宫格";
  updateVideoOverlayMode(referenceDevice.id);
  quadVideoDeviceIds().forEach((deviceId) => {
    const device = pickDevice(deviceId, isSelectableVideoDevice);
    if (device) ensureVideoElement(device, VIDEO_ROLE_MAIN);
  });
  wrap.dataset.videoLayout = VIDEO_LAYOUT_QUAD;
  setActiveMainVideoElement(referenceDevice.id);
  renderQuadVideoLabels();
  renderVideoDeviceList();
  renderAuxVideoControls();
  syncVideoPlayers().catch((error) => {
    log("四宫格视频池同步失败", { error: error.message });
  });
  if (logSwitch) {
    log("已进入四宫格模式");
  }
}

function collapseOverlayDrawers() {
  document.querySelectorAll("[data-overlay-drawer]").forEach((drawer) => {
    drawer.dataset.open = "false";
  });
}

function renderQuadVideoLabels() {
  const wrap = document.getElementById("videoFrameWrap");
  if (!wrap) return;
  wrap.querySelectorAll(".quad-video-label").forEach((label) => label.remove());
  if (state.videoLayout !== VIDEO_LAYOUT_QUAD) return;
  quadVideoDeviceIds().forEach((deviceId) => {
    const device = pickDevice(deviceId);
    const label = document.createElement("div");
    label.className = "quad-video-label";
    label.dataset.quadDevice = deviceId;
    label.dataset.quadSlot = quadVideoSlot(deviceId);
    label.textContent = device?.name || deviceId;
    wrap.appendChild(label);
  });
}

// 根据当前视频切换叠加控制组件
function updateVideoOverlayMode(videoDeviceId) {
  const dogPanel = document.getElementById("dogPtzVideoPanel");
  const roofPanel = document.getElementById("roofPtzVideoPanel");
  const joystickOverlay = document.getElementById("videoJoystickOverlay");
  const robotOverlay = document.getElementById("robotVideoOverlay");

  const isDogMode = !videoDeviceId || videoDeviceId === "dog_ptz_1"
    || videoDeviceId.startsWith("m20_");

  if (isDogMode) {
    if (dogPanel) dogPanel.hidden = false;
    if (roofPanel) roofPanel.hidden = true;
    if (joystickOverlay) joystickOverlay.style.display = "";
    if (robotOverlay) robotOverlay.style.display = "";
  } else if (videoDeviceId === "roof_ptz_1") {
    if (dogPanel) dogPanel.hidden = true;
    if (roofPanel) roofPanel.hidden = false;
    if (joystickOverlay) joystickOverlay.style.display = "none";
    if (robotOverlay) robotOverlay.style.display = "none";
  }
}

// 车顶视频面板控制同步
function bindRoofVideoPanel() {
  const liftSpeed = document.getElementById("roofLiftSpeedInput");
  const liftHeightSlider = document.getElementById("roofLiftHeightSlider");

  liftSpeed?.addEventListener("input", () => {
    const speed = Number(liftSpeed.value);
    const drawerSpeed = document.getElementById("liftSpeedInput");
    if (drawerSpeed) drawerSpeed.value = String(speed);
    updateLiftSpeedLabels(speed);
  });
  liftSpeed?.addEventListener("change", async () => {
    await sendLiftCommand({ action: "set_speed", speed: Number(liftSpeed.value) });
  });

  liftHeightSlider?.addEventListener("input", () => {
    syncLiftTargetLabels();
    const drawer = document.getElementById("liftHeightSlider");
    if (drawer) drawer.value = liftHeightSlider.value;
  });
  liftHeightSlider?.addEventListener("change", () => {

async function startWebRtcPlayer(device, { role = VIDEO_ROLE_MAIN } = {}) {
  const key = videoPlayerKey(role, device.id);
  const existing = state.videoPlayers.get(key);
  if (existing && existing.playerUrl === device.video.player_url && existing.status === "connected") {
    return existing;
  }
  if (existing && existing.status === "connecting") {
    return existing;
  }

  stopWebRtcPlayer(device.id, {
    role,
    allowReconnect: false,
    removeElement: false,
    clearVideo: role !== VIDEO_ROLE_AUX && state.videoLayout !== VIDEO_LAYOUT_QUAD,
  });

  const video = ensureVideoElement(device, role);
  const playerUrl = device.video.player_url;
  const whepUrl = `${playerUrl.replace(/\/$/, "")}/whep`;
  const pc = new RTCPeerConnection();
  const generation = (existing?.generation || 0) + 1;
  const player = {
    pc,
    resourceUrl: null,
    playerUrl,
    deviceId: device.id,
    role,
    generation,
    reconnectAttempt: existing?.reconnectAttempt || 0,
    reconnectTimer: null,
    status: "connecting",
    video,
  };
  state.videoPlayers.set(key, player);

  pc.addTransceiver("video", { direction: "recvonly" });
  pc.ontrack = (event) => {
    video.srcObject = event.streams[0];
    video.play().catch(() => {});
  };
  pc.onconnectionstatechange = () => {
    if (["failed", "disconnected", "closed"].includes(pc.connectionState)) {
      scheduleVideoReconnect(device.id, `connection ${pc.connectionState}`, generation, role);
    }
  };
  pc.oniceconnectionstatechange = () => {
    if (["failed", "disconnected"].includes(pc.iceConnectionState)) {
      scheduleVideoReconnect(device.id, `ice ${pc.iceConnectionState}`, generation, role);
    }
  };

  try {
    const offer = await pc.createOffer();
    await pc.setLocalDescription(offer);
    await waitForIceGathering(pc);

    const response = await fetch(whepUrl, {
      method: "POST",
      headers: { "Content-Type": "application/sdp" },
      body: pc.localDescription.sdp,
    });
    if (!response.ok) {
      throw new Error(`WHEP ${response.status}`);
    }

    const resourceUrl = response.headers.get("Location");
    player.resourceUrl = resourceUrl ? new URL(resourceUrl, whepUrl).toString() : null;
    const answer = await response.text();
    await pc.setRemoteDescription({ type: "answer", sdp: answer });
    player.reconnectAttempt = 0;
    player.status = "connected";
    renderVideoDeviceList();
    renderAuxVideoControls();
    updateStatusUi();
    log("WebRTC 视频已连接", { role, device_id: device.id, name: device.name, whep_url: whepUrl });
    return player;
  } catch (error) {
    player.status = "error";
    renderVideoDeviceList();
    renderAuxVideoControls();
    updateStatusUi();
    log("WebRTC 视频连接失败", { role, device_id: device.id, name: device.name, error: error.message, whep_url: whepUrl });
    scheduleVideoReconnect(device.id, error.message, generation, role);
    return player;
  }
}

function stopWebRtcPlayer(deviceId, { role = VIDEO_ROLE_MAIN, allowReconnect = false, removeElement = false, clearVideo = true } = {}) {
  const key = videoPlayerKey(role, deviceId);
  const player = state.videoPlayers.get(key);
  if (!player) return;

  if (!allowReconnect) {
    clearPlayerReconnect(deviceId, role);
  }
  player.generation += 1;
  state.videoPlayers.delete(key);
  if (player.video && clearVideo) {
    player.video.srcObject = null;
    player.video.dataset.active = "false";
    player.video.dataset.videoDevice = "";
    if (removeElement) {
      player.video.remove();
    }
  }
  if (player.resourceUrl) {
    fetch(player.resourceUrl, { method: "DELETE" }).catch(() => {});
  }
  player.pc.close();
}

  });
}

function scheduleVideoReconnect(deviceId, reason, generation, role = VIDEO_ROLE_MAIN) {
  const key = videoPlayerKey(role, deviceId);
  const player = state.videoPlayers.get(key);
  if (!player) return;
  if (generation !== player.generation) return;
  if (player.reconnectTimer) return;
  const device = pickDevice(deviceId);
  if (!device?.video?.player_url || device.video.enabled === false) return;

  const attempt = Math.min(player.reconnectAttempt + 1, 6);
  player.reconnectAttempt = attempt;
  const baseDelayMs = state.videoLayout === VIDEO_LAYOUT_QUAD && role === VIDEO_ROLE_MAIN ? 10000 : 1000;
  const delayMs = Math.min(30000, baseDelayMs * 2 ** (attempt - 1));
  log("WebRTC 视频将自动重连", { role, device_id: deviceId, reason, delay_ms: delayMs });
  player.reconnectTimer = setTimeout(async () => {
    const latest = state.videoPlayers.get(key);
    if (!latest || generation !== latest.generation) return;
    latest.reconnectTimer = null;
    stopWebRtcPlayer(deviceId, {
      role,
      allowReconnect: true,
      removeElement: false,
      clearVideo: !(state.videoLayout === VIDEO_LAYOUT_QUAD && role === VIDEO_ROLE_MAIN),
    });
    await startWebRtcPlayer(device, { role });
    if (role === VIDEO_ROLE_MAIN && state.videoLayout === VIDEO_LAYOUT_QUAD) {
      setActiveMainVideoElement(state.videoDevice?.id || "m20_front_camera");
      renderQuadVideoLabels();
      renderVideoDeviceList();
    } else if (role === VIDEO_ROLE_MAIN && state.videoDevice?.id === deviceId) {
      await activateVideoDevice(device, { logSwitch: false });
    } else if (role === VIDEO_ROLE_AUX && state.auxVideoDevice?.id === deviceId) {
      renderAuxVideoControls();
    }
  }, delayMs);
}

function clearPlayerReconnect(deviceId, role = VIDEO_ROLE_MAIN) {
  const player = state.videoPlayers.get(videoPlayerKey(role, deviceId));
  if (!player?.reconnectTimer) return;
  clearTimeout(player.reconnectTimer);
  player.reconnectTimer = null;
}

  if (!video) {
    video = wrap.querySelector('#videoFrame:not([data-video-device]), #videoFrame[data-video-device=""]') || document.createElement("video");

  return video;
}

function waitForIceGathering(pc) {
  if (pc.iceGatheringState === "complete") {
    return Promise.resolve();
  }
  return new Promise((resolve) => {
    const check = () => {
      if (pc.iceGatheringState === "complete") {
        pc.removeEventListener("icegatheringstatechange", check);
        resolve();
      }
    };
    pc.addEventListener("icegatheringstatechange", check);
  });
}

function connectControlSocket() {
  const protocol = location.protocol === "https:" ? "wss" : "ws";
  state.socket = new WebSocket(`${protocol}://${location.host}/ws/control?client_id=${encodeURIComponent(state.clientId)}`);

  state.socket.onopen = () => {
    document.getElementById("connectionStatus").textContent = "控制服务已连接";
    log("WebSocket connected");
  };

  state.socket.onmessage = async (event) => {
    const message = JSON.parse(event.data);
    if (message.type !== "device_status_updated" && message.type !== "lift_up" && message.type !== "lift_down") {
      log("事件", message);
    }
    if (message.type?.startsWith("lease_") && state.device && message.device_id === state.device.id) {
      await loadLease();
    }
    if (message.type?.startsWith("lease_") && state.liftDevice && message.device_id === state.liftDevice.id) {
      state.liftLease = message.payload;
      updateLiftOccupancyUi();
    }
    if (message.type?.startsWith("lift_") && state.liftDevice && message.device_id === state.liftDevice.id) {
      const remoteState = message.payload?.payload?.state;
      if (remoteState) syncLiftRemoteControls(remoteState);
    }
    if (message.type === "device_status_updated" && state.device && message.device_id === state.device.id) {
      state.status = message.payload;
      updateStatusUi();
    }
    if (message.type === "device_status_updated" && state.robotDevice && message.device_id === state.robotDevice.id) {
      renderRobotStatus(message.payload);
    }
    if (message.type?.startsWith("lease_") && state.robotDevice && message.device_id === state.robotDevice.id) {
      const lease = message.payload || {};
      if (state.robotMove.active && lease.holder_id && lease.holder_id !== state.clientId) {
        await stopRobotMove({ reason: "robot lease changed" });
      }
    }
    if (message.type === "control_failed") {
      updateLastFailure(message.payload);
      await loadControlLogs();
    }
  };

  state.socket.onclose = () => {
    document.getElementById("connectionStatus").textContent = "控制服务已断开，正在重连...";
    stopRobotMove({ reason: "websocket disconnected" });
    setTimeout(connectControlSocket, 1000);
  };
}

async function readJson(response) {
  const text = await response.text();
  let payload = {};
  try {
    payload = text ? JSON.parse(text) : {};
  } catch {
    payload = { detail: text || `HTTP ${response.status}` };
  }
  if (!response.ok) {
    const detail = typeof payload.detail === "string" ? payload.detail : JSON.stringify(payload.detail);
    throw new Error(detail || "request failed");
  }
  return payload;
}

async function getJson(url) {
  const response = await fetch(url, {
    headers: { "X-Patrol-Client-Id": state.clientId },
  });
  return readJson(response);
}

async function postJson(url, body = {}, method = "POST") {
  const response = await fetch(url, {
    method,
    headers: {
      "Content-Type": "application/json",
      "X-Patrol-Client-Id": state.clientId,
    },
    body: JSON.stringify(body),
  });
  return readJson(response);
}

  const battery = details.battery || {};
  const lights = details.lights || {};
  const cameraPower = details.camera_power || {};
  const motion = details.motion || {};
  const m20Motion = toM20AxisIntent(motion);
  const modeLabel = status.control_mode === "real" ? "真实" : "Dry-run";
  renderRobotOverlayStatus(status, details, battery, lights, cameraPower, motion, unknownRobotState);
  renderRobotFaults(details.faults || []);
  updateRobotControlStates(status, details, lights, unknownRobotState);
  if (unknownRobotState) {
    target.textContent = `${modeLabel} / 设备离线或尚未收到机器狗状态上报`;
    return;
  }
  target.textContent = [

function isRobotStanding(posture) {
  return posture === "stand" || posture === "rl_control";
}

function resolveRobotAction(action) {
  if (action === "posture_toggle") {
    const posture = state.robotStatus?.details?.posture;
    return isRobotStanding(posture) ? "lie_down" : "stand";
  }
  if (action === "sleep_toggle") {
    const sleep = state.robotStatus?.details?.sleep;
    return sleep === "awake" ? "sleep" : "wake";
  }
  return action;
}

function nextRobotLightTarget(lights = {}) {
  const front = Boolean(lights.front);
  const rear = Boolean(lights.rear);
  return front && rear
    ? { front: false, rear: false }
    : front
      ? { front: false, rear: true }
      : rear
        ? { front: true, rear: true }
        : { front: true, rear: false };
}

function isRobotMoving(motion = {}) {
  return Math.abs(Number(motion.x || 0)) > 0.01
    || Math.abs(Number(motion.y || 0)) > 0.01
    || Math.abs(Number(motion.rotation || 0)) > 0.01;
}

function isRobotVelocityMoving(velocity = {}) {
  return Math.abs(Number(velocity.x || 0)) > 0.02
    || Math.abs(Number(velocity.y || 0)) > 0.02
    || Math.abs(Number(velocity.yaw || 0)) > 0.02;
}

function resolveRobotVelocity(details = {}, motion = {}) {
  const velocity = details.actual_velocity;
  if (velocity && ["x", "y", "yaw"].some((axis) => velocity[axis] !== undefined && velocity[axis] !== null)) {
    return {
      x: Number(velocity.x || 0),
      y: Number(velocity.y || 0),
      yaw: Number(velocity.yaw || 0),
      source: velocity.source || "actual",
    };
  }
  const intent = toM20AxisIntent(motion);
  return {
    x: Number(intent.x || 0),
    y: Number(intent.y || 0),
    yaw: Number(intent.yaw || 0),
    source: "intent",
  };
}

function resolveRobotMoveIntentVelocity() {
  const movement = composeRobotMovement();
  if (!movement) return null;
  const speed = Math.max(Number(document.getElementById("robotSpeedInput")?.value || state.robotStatus?.details?.speed || 0.3), 0);
  const scaled = {
    x: (movement.x || 0) * speed,
    y: (movement.y || 0) * speed * (movement.y > 0 ? 1.4 : 1),
    yaw: (movement.rotation || 0) * speed,
  };
  const intent = toM20AxisIntent(scaled);
  return {
    x: intent.x,
    y: intent.y,
    yaw: intent.yaw,
    source: "command",
  };
}

function robotVelocityHtml(velocity) {
  return `
    <span class="robot-velocity-row"><em>X</em><b>${formatLinearVelocity(velocity.x)}</b><i>m/s</i></span>
    <span class="robot-velocity-row"><em>Y</em><b>${formatLinearVelocity(velocity.y)}</b><i>m/s</i></span>
    <span class="robot-velocity-row"><em>Yaw</em><b>${formatAngularVelocity(velocity.yaw)}</b><i>rad/s</i></span>
  `;
}

function robotCameraText(cameraPower = {}) {
  const frontKnown = Object.prototype.hasOwnProperty.call(cameraPower, "front");
  const rearKnown = Object.prototype.hasOwnProperty.call(cameraPower, "rear");
  if (!frontKnown && !rearKnown) return "未知";
  return `${frontKnown ? (cameraPower.front ? "前开" : "前关") : "前未知"} / ${rearKnown ? (cameraPower.rear ? "后开" : "后关") : "后未知"}`;
}

function formatAxis(value) {
  return Number(value || 0).toFixed(2);
}

function formatRobotPosition(details = {}) {
  const mapPose = details.map_pose || {};
  const gps = details.gps || {};
  const parts = [];
  if (Number.isFinite(Number(mapPose.x)) || Number.isFinite(Number(mapPose.y))) {
    const loc = Number(mapPose.location);
    const locText = loc === 0 ? "定位正常" : (loc === 1 ? "定位丢失" : "地图");
    parts.push(`${locText} X:${Number(mapPose.x || 0).toFixed(2)} Y:${Number(mapPose.y || 0).toFixed(2)} m`);
  }
  if (Number.isFinite(Number(gps.latitude)) || Number.isFinite(Number(gps.longitude))) {
    const sats = gps.num_satellites ?? gps.visible_satellites;
    parts.push(`GPS ${Number(gps.latitude || 0).toFixed(6)}, ${Number(gps.longitude || 0).toFixed(6)}${sats !== undefined && sats !== null ? ` · ${sats}星` : ""}`);
  }
  return parts.length ? parts.join(" | ") : "未知";
}

function robotPositionStatus(details = {}) {
  const mapPose = details.map_pose || {};
  if (Number(mapPose.location) === 1) return "warn";
  if (Number.isFinite(Number(mapPose.x)) || Number.isFinite(Number(details.gps?.latitude))) return "ok";
  return "unknown";
}

function formatLinearVelocity(value) {
  return Number(value || 0).toFixed(1);
}

function formatAngularVelocity(value) {
  return Number(value || 0).toFixed(1);
}

function toM20AxisIntent(movement = {}) {
  return {
    x: movement?.y || 0,
    y: -(movement?.x || 0),
    yaw: movement?.rotation || 0,
  };
}

function formatLiftDriverStatus(details, status) {
  const flags = details.status_flags || {};
  const statusText = details.status_text || "";
  const alarmText = details.alarm_text || "";
  const faultText = Array.isArray(details.faults) && details.faults.length ? details.faults.join(",") : "";
  if (alarmText) return alarmText;
  if (faultText) return faultText;
  if (details.motion === "up") {
    return statusText ? `上升中 · ${statusText}` : "上升中";
  }
  if (details.motion === "down") {
    return statusText ? `下降中 · ${statusText}` : "下降中";
  }
  if (flags.bit0 || statusText.includes("位置到达")) return statusText || "到位";
  if (statusText) return statusText;
  if (!status.online) return "离线";
  return "已停止";
}

async function loadLiftStatus() {
  const target = document.getElementById("liftStatus");
  if (!target || !state.liftDevice) return;
  try {
    const status = await getJson(`/api/devices/${state.liftDevice.id}/lift/status`);
    const details = status.details || {};
    const limits = details.limits || {};
    const driverStatus = formatLiftDriverStatus(details, status);
    const motorRpm = details.speed_rpm ?? details.motor_rpm ?? details.rpm ?? null;
    const hasHeight = details.height !== null && details.height !== undefined && Number.isFinite(Number(details.height));
    const heightPercent = hasHeight ? Number(details.height) : null;
    const heightCm = Number.isFinite(Number(details.height_cm))
      ? Number(details.height_cm)
      : (heightPercent === null ? null : (heightPercent / 100) * liftTravelCm());
    // total_height_cm 未配置时为 null；勿用 Number.isFinite(Number(null))（会得到 0）。
    const totalHeightCm = (
      details.total_height_cm !== null
      && details.total_height_cm !== undefined
      && details.total_height_cm !== ""
      && Number.isFinite(Number(details.total_height_cm))
    ) ? Number(details.total_height_cm) : null;
    if (Object.prototype.hasOwnProperty.call(details, "base_height_cm")) {
      if (!state.config) state.config = {};
      const nextBase = details.base_height_cm;
      if (state.config.lift_base_height_cm !== nextBase) {
        state.config.lift_base_height_cm = nextBase;
        applyLiftTravelToControls();
      } else {
        state.config.lift_base_height_cm = nextBase;
      }
    }
    const commandSpeed = Number.isFinite(Number(details.command_speed_cm_s))
      ? Number(details.command_speed_cm_s)
      : liftCommandSpeedCmS(details.speed);
    if (Number.isFinite(Number(details.travel_height_cm)) && Number(details.travel_height_cm) > 0) {
      if (!state.config) state.config = {};
      const nextTravel = Number(details.travel_height_cm);
      if (state.config.lift_travel_height !== nextTravel) {
        state.config.lift_travel_height = nextTravel;
        applyLiftTravelToControls();
      } else {
        state.config.lift_travel_height = nextTravel;
      }
    }
    updateLiftCurrentHeightUi(heightCm, totalHeightCm);
    syncLiftRemoteControls(details);
    await loadLiftLease();
    target.textContent = [
      status.control_mode === "real" ? "Real" : "Dry-run",
      formatLiftHeightLabel(heightCm, totalHeightCm),
      `速度 ${Number(details.speed ?? 0).toFixed(2)} · ${commandSpeed.toFixed(2)} cm/s`,
      `状态 ${liftMotionLabel(details.motion)}`,
      `驱动器 ${driverStatus}`,
      `转速 ${motorRpm === null ? "未知" : `${Number(motorRpm).toFixed(0)} rpm`}`,
      Number.isFinite(Number(details.zero_pulse)) && Number(details.zero_pulse) !== 0 ? "零点已标定" : "零点未标定",
      details.position_trusted === false ? "高度计数未可信" : null,
      limits.min ? "下限位" : null,
      limits.max ? "上限位" : null,
    ].filter(Boolean).join(" / ");
    renderLiftStatusFields(driverStatus, motorRpm, heightCm, totalHeightCm);
    scheduleLiftStatusPolling();
  } catch (error) {
    target.textContent = `状态读取失败：${error.message}`;
    renderLiftStatusFields("未知", null, null, null);
    scheduleLiftStatusPolling();
  }
}

function scheduleLiftStatusPolling() {
  if (state.liftStatusTimer) return;
  state.liftStatusTimer = setInterval(loadLiftStatus, 1000);
}

function bindLiftDragGuards() {
  const bind = (ids, flag) => {
    ids.forEach((id) => {
      const el = document.getElementById(id);
      if (!el || el.dataset.dragGuard === "1") return;
      el.dataset.dragGuard = "1";
      const start = () => { state[flag] = true; };
      el.addEventListener("pointerdown", start);
      el.addEventListener("touchstart", start, { passive: true });
      el.addEventListener("input", start);
    });
  };
  bind(["roofLiftHeightSlider", "liftHeightSlider"], "liftTargetDragging");
  bind(["roofLiftSpeedInput", "liftSpeedInput"], "liftSpeedDragging");
  if (document.documentElement.dataset.liftDragRelease !== "1") {
    document.documentElement.dataset.liftDragRelease = "1";
    const end = () => {
      state.liftTargetDragging = false;
      state.liftSpeedDragging = false;
    };
    document.addEventListener("pointerup", end);
    document.addEventListener("pointercancel", end);
    document.addEventListener("touchend", end, { passive: true });
  }
}

function syncLiftRemoteControls(details) {
  if (!details) return;
  if (!state.liftSpeedDragging) {
    const speed = Number(details.speed);
    if (Number.isFinite(speed)) {
      const clamped = Math.max(0, Math.min(1, speed));
      if (state.lastRemoteLiftSpeed !== clamped) {
        state.lastRemoteLiftSpeed = clamped;
        const roof = document.getElementById("roofLiftSpeedInput");
        const drawer = document.getElementById("liftSpeedInput");
        if (roof) roof.value = String(clamped);
        if (drawer) drawer.value = String(clamped);
        updateLiftSpeedLabels(clamped);
      }
    }
  }
  if (state.liftTargetDragging) return;
  const targetPct = details.target_height;
  if (targetPct === null || targetPct === undefined || targetPct === "") return;
  const pct = Number(targetPct);
  if (!Number.isFinite(pct)) return;
  if (state.lastRemoteLiftTargetPct === pct) return;
  state.lastRemoteLiftTargetPct = pct;
  const sliderVal = String(liftExtensionToSliderValue((pct / 100) * liftTravelCm()));
  const roofH = document.getElementById("roofLiftHeightSlider");
  const drawerH = document.getElementById("liftHeightSlider");
  if (roofH) roofH.value = sliderVal;
  if (drawerH) drawerH.value = sliderVal;
  syncLiftTargetLabels();
}

function renderControlLogs() {
  const target = document.getElementById("controlLogs");
  if (!target) return;
  if (!state.logs.length) {
    target.innerHTML = '<p class="empty-log">暂无控制日志</p>';
    updateLastFailure(null);
    return;
  }

  target.innerHTML = state.logs.slice(0, 30).map((entry) => {
    const time = new Date(entry.timestamp).toLocaleTimeString();
    const status = entry.ok ? "成功" : "失败";
    const detail = entry.ok
      ? JSON.stringify(entry.payload || {})
      : `${entry.category || "unknown"} ${entry.error || ""}`;
    return `
      <div class="log-item" data-ok="${entry.ok}">
        <div>
          <strong>${time} ${entry.action}</strong>
          <span>${entry.client_id || "unknown"} / ${entry.dry_run ? "dry-run" : "real"}</span>
        </div>
        <em>${status}</em>
        <p>${escapeHtml(detail)}</p>
      </div>
    `;
  }).join("");
}

function escapeHtml(value) {
  return String(value)
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}

function cssEscape(value) {
  if (window.CSS?.escape) {
    return CSS.escape(value);
  }
  return String(value).replaceAll('"', '\\"');
}

}

function updateStatusUi() {
  const status = state.status;
  if (!status) return;
  // 紧凑状态圆点（抽屉 + 视频面板）
  const setDot = (id, value, key = id) => {
    const dot = document.getElementById(id);
    if (dot) dot.dataset.status = stableStatus(`dot:${key}`, value || "unknown");
  };
  setDot("dotOnline", status.online ? "ok" : "error");
  // 视频状态优先用 WebRTC 播放器实际连接状态
  let videoStatus = status.video;
  const videoDeviceId = state.videoDevice?.id;
  if (videoDeviceId) {
    const player = state.videoPlayers.get(videoPlayerKey(VIDEO_ROLE_MAIN, videoDeviceId));
    if (player?.status === "connected") videoStatus = "ok";
    else if (player?.status === "connecting") videoStatus = "connecting";
    else if (player?.status === "error") videoStatus = "error";
  }
  setDot("dotVideo", videoStatus, `dog:video:${videoDeviceId || "none"}`);
  setDot("dotOnvif", status.onvif);
  setDot("dotIsapi", status.isapi);
  // 视频面板指示灯（与抽屉同步状态）
  const videoDots = document.querySelectorAll("#dogPtzVideoDots .status-dot");
  const map2 = {
    online: status.online ? "ok" : "error",
    video: videoStatus,
    onvif: status.onvif || "unknown",
    isapi: status.isapi || "unknown",
  };
  videoDots.forEach((dot) => {
    const key = dot.dataset.dot;
    if (key && map2[key] !== undefined) {
      dot.dataset.status = stableStatus(`video-dot:dog:${key}`, map2[key]);
    }
  });
}

async function acquireLease() {
  if (!state.device) return;
  const holderName = `操作员 ${state.clientId.slice(-4)}`;
  document.querySelectorAll('[data-ptz-device="dog_ptz_1"] [data-ptz-lease]').forEach((btn) => { btn.disabled = true; });
  try {
    state.lease = await postJson(`/api/devices/${state.device.id}/lease/acquire`, { holder_name: holderName });
    updateLeaseUi();
    log("已获取控制权", state.lease);
  } catch (error) {
    log("获取控制权失败", { error: error.message });
    await loadLease();
  } finally {
    updateLeaseUi();
  }
}

async function renewLease() {
  if (!state.device || !state.lease?.can_control) return;
  try {
    state.lease = await postJson(`/api/devices/${state.device.id}/lease/renew`);
    updateLeaseUi();
  } catch (error) {
    log("控制权续租失败", { error: error.message });
    await loadLease();
  }
}

async function releaseLease({ quiet = false } = {}) {
  if (!state.device || !state.lease?.can_control) return;
  try {
    state.lease = await postJson(`/api/devices/${state.device.id}/lease/release`);
    updateLeaseUi();
    if (!quiet) log("已释放控制权", state.lease);
  } catch (error) {
    if (!quiet) log("释放控制权失败", { error: error.message });
  }
}

function updateLeaseUi() {
  const lease = state.lease;
  let status, disabled;
  if (!lease || !lease.holder_id) {
    status = "none"; disabled = false;
  } else if (lease.can_control) {
    status = "self"; disabled = false;
  } else {
    status = "other"; disabled = true;
  }
  // 更新所有租约按钮（抽屉 + 视频面板）
  document.querySelectorAll('[data-ptz-device="dog_ptz_1"] [data-ptz-lease]').forEach((btn) => {
    btn.dataset.status = status;
    btn.textContent = "权";
    btn.disabled = disabled;
  });
  // 同步视频面板车顶租约按钮
  document.querySelectorAll('[data-roof-lease]').forEach((btn) => {
    const rl = state.roofPtzLease;
    let rs = !rl || !rl.holder_id ? "none" : rl.can_control ? "self" : "other";
    let rd = !rl || !rl.holder_id ? false : rl.can_control ? false : true;
    btn.dataset.status = rs;
    btn.disabled = rd;
  });

  const canControl = Boolean(lease?.can_control);
  if (!canControl) sendStopForActiveCommand();
  setControlButtonsEnabled(canControl);
  scheduleRenewal();
  scheduleLeasePolling();
}

function scheduleRenewal() {
  if (state.renewTimer) {
    clearInterval(state.renewTimer);
    state.renewTimer = null;
  }
  if (!state.lease?.can_control) return;

  const intervalMs = Math.max(3000, Math.floor((state.lease.ttl_seconds * 1000) / 2));
  state.renewTimer = setInterval(renewLease, intervalMs);
}

function scheduleLeasePolling() {
  if (state.leasePollTimer) return;
  state.leasePollTimer = setInterval(loadLease, 3000);
}

function startActiveRenewal() {
  stopActiveRenewal();
  state.activeRenewTimer = setInterval(renewLeaseForActiveCommand, 2000);
}

function stopActiveRenewal() {
  if (!state.activeRenewTimer) return;
  clearInterval(state.activeRenewTimer);
  state.activeRenewTimer = null;
}

// 车顶云台控制事件绑定
function bindRoofPtzControls() {
  // 控制权按钮
  roofPtzSectionQueryAll("[data-roof-lease]").forEach((acquireBtn) => acquireBtn.addEventListener("click", async () => {
    if (state.roofPtzLease?.can_control) { await releaseRoofPtzLease(); }
    else { await acquireRoofPtzLease(); }
  }));

  // 预置位1
  roofPtzSectionQueryAll("[data-preset-goto1]").forEach((button) => button.addEventListener("click", async () => {
    try {
      await roofPtzControlPost(`/api/devices/${ROOF_PTZ_ID}/presets/goto`, { preset: 1 });
      log("车顶云台调用预置位1");
    } catch (error) {
      log("车顶云台预置位1失败", { error: error.message });
    }
  }));

  // PTZ 方向/变倍按钮
  roofPtzSectionQueryAll("[data-command]").forEach((button) => {
    const command = button.dataset.command;
    const start = async (event) => {
      event.preventDefault();
      if (button.disabled || state.roofPtzActiveCommand) return;
      try {
        state.roofPtzActiveCommand = command;
        await roofPtzControlPost(`/api/devices/${ROOF_PTZ_ID}/ptz/start`, { command, speed: 5 });
      } catch (error) {
        state.roofPtzActiveCommand = null;
        log("车顶云台控制失败", { command, error: error.message });
      }
    };
    const stop = async (event) => {
      event.preventDefault();
      if (state.roofPtzActiveCommand !== command) return;
      state.roofPtzActiveCommand = null;
      try {
        await roofPtzControlPost(`/api/devices/${ROOF_PTZ_ID}/ptz/stop`, { command, speed: 5 });
      } catch (error) {
        log("车顶云台停止失败", { error: error.message });
      }
    };
    button.addEventListener("mousedown", start);
    button.addEventListener("touchstart", start, { passive: false });
    button.addEventListener("mouseup", stop);
    button.addEventListener("mouseleave", stop);
    button.addEventListener("touchend", stop, { passive: false });
    button.addEventListener("touchcancel", stop, { passive: false });
  });

  // 预置位按钮
  roofPtzSectionQueryAll("[data-preset-action]").forEach((button) => {
    button.addEventListener("click", async () => {
      const input = document.querySelector(`[data-ptz-device="${ROOF_PTZ_ID}"] .ptz-preset-input`);
      const preset = input ? Number(input.value) : 1;
      if (!preset) { log("请先选择预置位"); return; }
      try {
        await roofPtzControlPost(`/api/devices/${ROOF_PTZ_ID}/presets/${button.dataset.presetAction}`, { preset });
      } catch (error) {
        log("车顶预置位操作失败", { error: error.message });
      }
    });
  });

  // 预置位选择
  roofPtzSectionQueryAll("[data-preset-select]").forEach((button) => {
    button.addEventListener("click", () => {
      const active = button.dataset.active === "true";
      roofPtzSectionQueryAll("[data-preset-select]").forEach((item) => { item.dataset.active = "false"; });
      const input = document.querySelector(`[data-ptz-device="${ROOF_PTZ_ID}"] .ptz-preset-input`);
      if (active) { if (input) input.value = ""; return; }
      button.dataset.active = "true";
      if (input) input.value = button.dataset.presetSelect;
    });
  });

  // 补光灯
  roofPtzSectionQueryAll("[data-light-toggle]").forEach((button) => {
    button.addEventListener("click", async () => {
      if (button.disabled) return;
      const enabled = !state.roofPtzLightOn;
      try {
        await roofPtzControlPost(`/api/devices/${ROOF_PTZ_ID}/light`, { enabled });
        state.roofPtzLightOn = enabled;
        updateRoofPtzLightButtonUI();
        log("车顶补光灯已" + (enabled ? "打开" : "关闭"));
      } catch (error) {
        log("车顶补光灯控制失败", { error: error.message });
      }
    });
  });
}

// 视频区悬浮摇杆管理
const VIDEO_JOYSTICK_HIDE_DELAY = 1500;
let videoJoystickHideTimer = null;
let videoJoystickActive = false;

function showVideoJoysticks() {
  const overlay = document.getElementById("videoJoystickOverlay");
  if (!overlay) return;
  videoJoystickActive = true;
  overlay.classList.add("active");
  clearVideoJoystickHideTimer();
}

function scheduleVideoJoystickHide() {
  clearVideoJoystickHideTimer();
  videoJoystickHideTimer = setTimeout(() => {
    if (!state.robotMove.activeInputs.size) {
      const overlay = document.getElementById("videoJoystickOverlay");
      if (overlay) overlay.classList.remove("active");
      videoJoystickActive = false;
    }
  }, VIDEO_JOYSTICK_HIDE_DELAY);
}

function clearVideoJoystickHideTimer() {
  if (videoJoystickHideTimer) {
    clearTimeout(videoJoystickHideTimer);
    videoJoystickHideTimer = null;
  }
}

// 键盘模式下强制显示摇杆
function updateVideoJoystickVisibility() {
  const overlay = document.getElementById("videoJoystickOverlay");
  if (!overlay) return;
  if (state.robotInputMode === "keyboard") {
    overlay.classList.add("active");
    videoJoystickActive = true;
    clearVideoJoystickHideTimer();
  } else if (!state.robotMove.activeInputs.size) {
    overlay.classList.remove("active");
    videoJoystickActive = false;
  }
}

function bindRobotJoysticks() {
  // 视频区悬浮摇杆
  const overlay = document.getElementById("videoJoystickOverlay");
  if (overlay) {
    // 触摸检测：pointermove 在视频区底部时显示摇杆
    const videoWrap = document.getElementById("videoFrameWrap");
    if (videoWrap) {
      videoWrap.addEventListener("pointermove", (event) => {
        const rect = videoWrap.getBoundingClientRect();
        const relY = (event.clientY - rect.top) / rect.height;
        // 手指在底部 35% 区域时显示摇杆
        if (relY > 0.65) {
          showVideoJoysticks();
        } else if (videoJoystickActive && !state.robotMove.activeInputs.size) {
          scheduleVideoJoystickHide();
        }
      });
      videoWrap.addEventListener("pointerleave", () => {
        if (!state.robotMove.activeInputs.size) scheduleVideoJoystickHide();
      });
    }

    // 为悬浮摇杆绑定 pointer 事件
    overlay.querySelectorAll(".video-joystick-zone").forEach((zone) => {
      const joystickType = zone.dataset.zone === "left" ? "translate" : "rotate";
      const inputId = `joystick-${joystickType}`;
      const pad = zone.querySelector(".joystick-pad");
      const knob = zone.querySelector(".joystick-knob");

      const start = (event) => {
        if (state.robotInputMode !== "joystick") return;
        event.preventDefault();
        showVideoJoysticks();
        zone.setPointerCapture(event.pointerId);
        updateOverlayJoystickInput(zone, pad, knob, event, inputId, joystickType);
      };
      const move = (event) => {
        if (state.robotInputMode !== "joystick") return;
        if (!zone.hasPointerCapture(event.pointerId)) return;
        event.preventDefault();
        updateOverlayJoystickInput(zone, pad, knob, event, inputId, joystickType);
      };
      const stop = (event) => {
        event.preventDefault();
        if (zone.hasPointerCapture(event.pointerId)) {
          zone.releasePointerCapture(event.pointerId);
        }
        resetOverlayJoystickKnob(knob);
        stopRobotMoveInput(inputId, { reason: "joystick released" });
        scheduleVideoJoystickHide();
      };
      zone.addEventListener("pointerdown", start);
      zone.addEventListener("pointermove", move);
      zone.addEventListener("pointerup", stop);
      zone.addEventListener("pointercancel", stop);
      zone.addEventListener("lostpointercapture", () => {
        resetOverlayJoystickKnob(knob);
        stopRobotMoveInput(inputId, { reason: "joystick capture lost" });
        scheduleVideoJoystickHide();
      });
    });

  }

  // 旧的右侧面板摇杆保留兼容（display:none 时不会触发）
  document.querySelectorAll("[data-robot-joystick]").forEach((joystick) => {
    const inputId = `joystick-${joystick.dataset.robotJoystick}`;
    const start = (event) => {
      if (state.robotInputMode !== "joystick") return;
      event.preventDefault();
      joystick.setPointerCapture(event.pointerId);
      updateRobotJoystickInput(joystick, event, inputId);
    };
    const move = (event) => {
      if (state.robotInputMode !== "joystick") return;
      if (!joystick.hasPointerCapture(event.pointerId)) return;
      event.preventDefault();
      updateRobotJoystickInput(joystick, event, inputId);
    };
    const stop = (event) => {
      event.preventDefault();
      if (joystick.hasPointerCapture(event.pointerId)) {
        joystick.releasePointerCapture(event.pointerId);
      }
      resetRobotJoystick(joystick);
      stopRobotMoveInput(inputId, { reason: "joystick released" });
    };
    joystick.addEventListener("pointerdown", start);
    joystick.addEventListener("pointermove", move);
    joystick.addEventListener("pointerup", stop);
    joystick.addEventListener("pointercancel", stop);
    joystick.addEventListener("lostpointercapture", () => {
      resetRobotJoystick(joystick);
      stopRobotMoveInput(inputId, { reason: "joystick capture lost" });
    });
  });
}

function updateRobotJoystickInput(joystick, event, inputId) {
  const pad = joystick.querySelector(".joystick-pad");
  const knob = joystick.querySelector(".joystick-knob");
  const rect = pad.getBoundingClientRect();
  const radius = rect.width / 2;
  const centerX = rect.left + radius;
  const centerY = rect.top + radius;
  let dx = (event.clientX - centerX) / radius;
  let dy = (event.clientY - centerY) / radius;
  const length = Math.hypot(dx, dy);
  if (length > 1) {
    dx /= length;
    dy /= length;
  }
  setRobotJoystickKnob(joystick, dx, dy);

  // The payload still uses the legacy frontend field names; render and protocol
  // conversion expose them as M20 X/Y/Yaw so operators see the robot coordinate system.
  const movement = joystick.dataset.robotJoystick === "rotate"
    ? { x: 0, y: 0, rotation: applyJoystickDeadzone(dx) }
    : { x: applyJoystickDeadzone(dx), y: applyJoystickDeadzone(-dy), rotation: 0 };
  startRobotMoveInput(inputId, movement);
}

function applyJoystickDeadzone(value) {
  return Math.abs(value) < 0.08 ? 0 : Number(value.toFixed(2));
}

// 悬浮摇杆输入处理
function updateOverlayJoystickInput(zone, pad, knob, event, inputId, joystickType) {
  const rect = pad.getBoundingClientRect();
  const radius = rect.width / 2;
  const centerX = rect.left + radius;
  const centerY = rect.top + radius;
  let dx = (event.clientX - centerX) / radius;
  let dy = (event.clientY - centerY) / radius;
  const length = Math.hypot(dx, dy);
  if (length > 1) { dx /= length; dy /= length; }
  setJoystickKnobByDeltas(knob, dx, dy);

  const movement = joystickType === "rotate"
    ? { x: 0, y: 0, rotation: applyJoystickDeadzone(dx) }
    : { x: applyJoystickDeadzone(dx), y: applyJoystickDeadzone(-dy), rotation: 0 };
  startRobotMoveInput(inputId, movement);
}

function resetOverlayJoystickKnob(knob) {
  if (knob) knob.style.transform = "translate(0px, 0px)";
}

function setJoystickKnobByDeltas(knob, dx, dy) {
  if (!knob) return;
  const pad = knob.parentElement;
  const radius = pad.getBoundingClientRect().width / 2;
  const knobRange = Math.max(0, radius - knob.offsetWidth / 2 - 4);
  knob.style.transform = `translate(${clampAxis(dx) * knobRange}px, ${clampAxis(dy) * knobRange}px)`;
}

function resetRobotJoystick(joystick) {
  setRobotJoystickKnob(joystick, 0, 0);
}

function setRobotJoystickKnob(joystick, dx, dy) {
  const pad = joystick.querySelector(".joystick-pad");
  const knob = joystick.querySelector(".joystick-knob");
  if (!pad || !knob) return;
  const radius = pad.getBoundingClientRect().width / 2;
  const knobRange = Math.max(0, radius - knob.offsetWidth / 2 - 4);
  knob.style.transform = `translate(${clampAxis(dx) * knobRange}px, ${clampAxis(dy) * knobRange}px)`;
}

function isEditableTarget(target) {
  if (!target) return false;
  const tag = target.tagName;
  return tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || target.isContentEditable;
}

async function sendRobotCommand(command) {
  if (!state.robotDevice) return null;
  if (command.action === "stop_move") {
    await stopRobotMove({ reason: "manual stop" });
    return null;
  }
  if (command.action !== "move") {
    await stopRobotMove({ reason: `robot ${command.action}`, quiet: true });
  }
  if (command.action !== "soft_estop") {
    await ensureRobotLease();
  }
  const result = await postJson(`/api/devices/${state.robotDevice.id}/robot/control`, command);
  const operatorDirection = result.payload?.state?.operator_direction;
  if (command.action === "direction_reverse" && operatorDirection) {
    state.operatorDirection = operatorDirection;
    await switchVideoForRobotDirection(operatorDirection);
    if (state.videoLayout === VIDEO_LAYOUT_QUAD) {
      quadVideoDeviceIds().forEach((deviceId) => {
        const video = document.querySelector(`.video-frame[data-video-role="main"][data-video-device="${CSS.escape(deviceId)}"]`);
        if (video) video.dataset.quadSlot = quadVideoSlot(deviceId);
      });
      renderQuadVideoLabels();
    }
    renderVideoDeviceList();
  }
  scheduleRobotStatusBurst();
  return result;
}

async function startRobotMoveInput(inputId, movement) {
  state.robotMove.activeInputs.set(inputId, movement);
  await updateRobotMoveFromInputs();
}

async function stopRobotMoveInput(inputId, { reason = "input released" } = {}) {
  state.robotMove.activeInputs.delete(inputId);
  await updateRobotMoveFromInputs({ reason });
}

async function updateRobotMoveFromInputs({ reason = "inputs changed" } = {}) {
  const movement = composeRobotMovement();
  renderRobotMoveIntent(movement);
  renderRobotJoystickFeedback(movement);
  if (!movement) {
    await stopRobotMove({ reason });
    return;
  }
  await startRobotMove(movement);
}

function composeRobotMovement() {
  if (!state.robotMove.activeInputs.size) return null;
  const merged = { x: 0, y: 0, rotation: 0 };
  state.robotMove.activeInputs.forEach((movement) => {
    merged.x += movement.x || 0;
    merged.y += movement.y || 0;
    merged.rotation += movement.rotation || 0;
  });
  merged.x = clampAxis(merged.x);
  merged.y = clampAxis(merged.y);
  merged.rotation = clampAxis(merged.rotation);
  if (!merged.x && !merged.y && !merged.rotation) return null;
  return merged;
}

function clampAxis(value) {
  return Math.max(-1, Math.min(1, value));
}

function renderRobotMoveIntent(movement, velocity = null, unknownRobotState = false) {
  const target = document.getElementById("robotMoveIntent");
  if (!target) return;
  const actual = velocity || resolveRobotVelocity(state.robotStatus?.details || {}, movement || {});
  const command = state.robotMove.active ? resolveRobotMoveIntentVelocity() : null;
  if (unknownRobotState) {
    target.textContent = "实际 X:未知 Y:未知 Yaw:未知";
    return;
  }
  if (command) {
    target.textContent = `指令 X:${formatLinearVelocity(command.x)}m/s Y:${formatLinearVelocity(command.y)}m/s Yaw:${formatAngularVelocity(command.yaw)}rad/s · 实际 X:${formatLinearVelocity(actual.x)}m/s Y:${formatLinearVelocity(actual.y)}m/s Yaw:${formatAngularVelocity(actual.yaw)}rad/s`;
    return;
  }
  target.textContent = `实际 X:${formatLinearVelocity(actual.x)}m/s Y:${formatLinearVelocity(actual.y)}m/s Yaw:${formatAngularVelocity(actual.yaw)}rad/s`;
}

function renderRobotJoystickFeedback(movement) {
  // 视频区悬浮摇杆
  const overlay = document.getElementById("videoJoystickOverlay");
  if (overlay) {
    const leftZone = overlay.querySelector('[data-zone="left"]');
    const rightZone = overlay.querySelector('[data-zone="right"]');
    if (leftZone) {
      const knob = leftZone.querySelector(".joystick-knob");
      setJoystickKnobByDeltas(knob, movement?.x || 0, -(movement?.y || 0));
    }
    if (rightZone) {
      const knob = rightZone.querySelector(".joystick-knob");
      setJoystickKnobByDeltas(knob, movement?.rotation || 0, 0);
    }
  }
  // 旧摇杆兼容（已隐藏但保留逻辑）
  const translate = document.querySelector('[data-robot-joystick="translate"]');
  const rotate = document.querySelector('[data-robot-joystick="rotate"]');
  if (translate) setRobotJoystickKnob(translate, movement?.x || 0, -(movement?.y || 0));
  if (rotate) setRobotJoystickKnob(rotate, movement?.rotation || 0, 0);
}

function resetAllRobotJoysticks() {
  renderRobotJoystickFeedback(null);
}

async function startRobotMove(movement) {
  if (!state.robotDevice) return;
  const speedInput = document.getElementById("robotSpeedInput");
  const speed = Number(speedInput?.value ?? state.robotStatus?.details?.speed ?? 0.3);
  if (speed <= 0) {
    log("机器狗速度为 0，请调高右侧速度滑条后再移动");
    return;
  }
  if (!state.robotMove.active) {
    try {
      await ensureRobotLease();
    } catch (error) {
      log("机器狗控制权获取失败", { error: error.message });
      return;
    }

    state.robotMove.active = true;
    state.robotMove.timer = setInterval(sendRobotMoveTick, ROBOT_MOVE_INTENT_INTERVAL_MS);
    state.robotMove.renewTimer = setInterval(() => {
      ensureRobotLease().catch((error) => {
        log("机器狗控制权续租失败", { error: error.message });
        stopRobotMove({ reason: "robot lease renew failed" });
      });
    }, 2000);
  }

  state.robotMove.command = { action: "move", ...movement };
  await sendRobotMoveTick();
}

async function sendRobotMoveTick() {
  if (!state.robotMove.active || !state.robotMove.command || state.robotMove.inFlight || !state.robotDevice) return;
  state.robotMove.inFlight = true;
  try {
    await postJson(`/api/devices/${state.robotDevice.id}/robot/control`, state.robotMove.command);
    if (state.robotMove.active) {
      renderRobotMoveIntent(state.robotMove.command, resolveRobotVelocity(state.robotStatus?.details || {}));
    }
  } catch (error) {
    log("机器狗运动控制失败", { error: error.message });
    await stopRobotMove({ reason: "robot move failed", quiet: true, waitForInFlight: false });
  } finally {
    state.robotMove.inFlight = false;
  }
}

async function stopRobotMove({ reason = "stop", quiet = false, waitForInFlight = true } = {}) {
  const wasActive = state.robotMove.active || state.robotMove.command;
  state.robotMove.activeInputs.clear();
  resetAllRobotJoysticks();
  renderRobotMoveIntent(null);
  clearRobotMoveTimers();
  state.robotMove.active = false;
  state.robotMove.command = null;
  if (!state.robotDevice || state.robotMove.stopInFlight) return;
  if (!wasActive && quiet) return;

  if (waitForInFlight) {
    await waitForRobotMoveIdle();
  }

  state.robotMove.stopInFlight = true;
  try {
    await postJson(`/api/devices/${state.robotDevice.id}/robot/control`, { action: "stop_move" });
    if (!quiet) log("机器狗运动已归零", { reason });
    scheduleRobotStatusBurst();
  } catch (error) {
    if (!quiet) log("机器狗归零失败", { reason, error: error.message });
  } finally {
    state.robotMove.stopInFlight = false;
  }
}

async function waitForRobotMoveIdle(timeoutMs = 800) {
  const startedAt = Date.now();
  while (state.robotMove.inFlight && Date.now() - startedAt < timeoutMs) {
    await new Promise((resolve) => setTimeout(resolve, 10));
  }
}

function clearRobotMoveTimers() {
  if (state.robotMove.timer) {
    clearInterval(state.robotMove.timer);
    state.robotMove.timer = null;
  }
  if (state.robotMove.renewTimer) {
    clearInterval(state.robotMove.renewTimer);
    state.robotMove.renewTimer = null;
  }
}

const liftHoldControllers = new Set();

function registerLiftHoldController(controller) {
  liftHoldControllers.add(controller);
}

function resetLiftHoldControllers() {
  liftHoldControllers.forEach((controller) => {
    controller.engaged = false;
    controller.token += 1;
    if (controller.keepaliveTimer) {
      clearTimeout(controller.keepaliveTimer);
      controller.keepaliveTimer = null;
    }
  });
}

function bindLiftHoldButton(button, action) {
  const controller = { engaged: false, starting: false, token: 0, keepaliveTimer: null };
  registerLiftHoldController(controller);

  const clearKeepalive = () => {
    if (controller.keepaliveTimer) {
      clearTimeout(controller.keepaliveTimer);
      controller.keepaliveTimer = null;
    }
  };

  const release = async (event) => {
    event?.preventDefault();
    if (!controller.engaged) return;
    controller.engaged = false;
    controller.token += 1;
    clearKeepalive();
    // 启动还在写寄存器时不要并发发 stop，否则 256 会插进使能/点动中间，杆几乎不动。
    if (controller.starting) return;
    try {
      await sendLiftCommand({ action: "stop" });
    } catch (error) {
      log("升降杆停止失败", { action, error: error.message });
    }
  };

  const engage = async (event) => {
    event.preventDefault();
    if (controller.engaged) return;
    if (event.type === "mousedown" && event.button !== 0) return;
    controller.engaged = true;
    const token = controller.token;
    controller.starting = true;
    try {
      await sendLiftCommand({ action });
    } catch (error) {
      if (token === controller.token) controller.engaged = false;
      clearKeepalive();
      log("升降杆点动启动失败", { action, error: error.message });
      return;
    } finally {
      controller.starting = false;
    }
    if (!controller.engaged || token !== controller.token) {
      try {
        await sendLiftCommand({ action: "stop" });
      } catch (error) {
        log("升降杆停止失败", { action, error: error.message });
      }
      return;
    }
    const tick = async () => {
      if (!controller.engaged || token !== controller.token) return;
      try {
        await sendLiftCommand({ action });
      } catch (error) {
        log("升降杆点动保持失败", { action, error: error.message });
      }
      if (controller.engaged && token === controller.token) {
        controller.keepaliveTimer = setTimeout(tick, LIFT_JOG_KEEPALIVE_MS);
      }
    };
    controller.keepaliveTimer = setTimeout(tick, LIFT_JOG_KEEPALIVE_MS);
  };

  button.addEventListener("mousedown", engage);
  button.addEventListener("touchstart", engage, { passive: false });
  button.addEventListener("mouseup", release);
  button.addEventListener("mouseleave", release);
  button.addEventListener("touchend", release, { passive: false });
  button.addEventListener("touchcancel", release, { passive: false });
}

}

function captureCurrentVideo() {
  const video = document.querySelector(".video-frame[data-active=\"true\"]");
  if (!video || !video.videoWidth || !video.videoHeight) {
    log("截图失败", { error: "当前视频尚未输出画面" });
    return;
  }
  const canvas = document.createElement("canvas");
  canvas.width = video.videoWidth;
  canvas.height = video.videoHeight;
  const context = canvas.getContext("2d");
  if (!context) {
    log("截图失败", { error: "浏览器不支持 Canvas 截图" });
    return;
  }
  try {
    context.drawImage(video, 0, 0, canvas.width, canvas.height);
    const link = document.createElement("a");
    const timestamp = new Date().toISOString().replace(/[:.]/g, "-");
    link.download = `patrol-${state.videoDevice?.id || "video"}-${timestamp}.png`;
    link.href = canvas.toDataURL("image/png");
    link.click();
    log("截图已下载", { device_id: state.videoDevice?.id, file: link.download });
  } catch (error) {
    log("截图失败", { error: error.message });
  }
}

function togglePresetSelection(button) {
  const section = button.closest("[data-ptz-device]");
  if (!section) return;
  const active = button.dataset.active === "true";
  section.querySelectorAll("[data-preset-select]").forEach((item) => {
    item.dataset.active = "false";
  });
  const input = section.querySelector(".ptz-preset-input");
  if (active) {
    input.value = "";
    return;
  }
  button.dataset.active = "true";
  input.value = button.dataset.presetSelect;
}

    method: "POST",

  }).catch(() => {});
}
